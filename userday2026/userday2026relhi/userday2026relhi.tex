\section{Finalized new GPR API migration}

\begin{frame}
\frametitle{Gaussian process regression API}

\begin{block}{Better hyperparameter optimization (1.26)}
\begin{itemize}
\item Normalization and improved scale bounds initialization
\item Faster and more robust convergence
\end{itemize}
\end{block}

\begin{block}{Heteroskedastic noise (1.27)}
\begin{itemize}
\item Input-dependent noise in GPR
\item New accessors: getCholeskyFactor, getStandardizedOutput
\end{itemize}
\end{block}

\begin{block}{Cross-validation (1.27 / experimental)}
\begin{itemize}
\item \texttt{GaussianProcessRegressionCrossValidation}
\item Automatic hyperparameter selection
\item Reworked GPR \& chaos documentation
\end{itemize}
\end{block}

\end{frame}

\section{Misc}

\begin{frame}
\frametitle{Documentation improvements}
\begin{itemize}
\item Improved functions API with graphs
\item Improved documentation of Directional sampling classes
\item Consistency of examples
\item SVG graphics
\item New example: pitfalls in polynomial chaos expansion
\item Improved Pearson and Spearman theory files
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}
\frametitle{Other improvements}
\begin{itemize}
\item Smolyak nested tensorization
\item Advanced validation of Distribution classes
\item Allow MultiStart optimization in FORM algorithms
\item Deepcopy support for all objects
\item FunctionalChaosResult.getMarginal
\item KDTree radius search
\item Mesh convex flag
\item TruncatedDistribution handles KernelMixture
\item Centered data in Linear/QuadraticLeastSquares
\item QuantileConfidence interval coverage
\item KFoldSplitter non-interlaced folds
\item Pagmo incremental evolution callback
\item SpecFunc: BesselInu/LogBesselInu/Clip
\end{itemize}
