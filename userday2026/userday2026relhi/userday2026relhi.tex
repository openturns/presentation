\begin{frame}[fragile]
\frametitle{Heteroscedastic noise in GPR}

\begin{columns}
\begin{column}{0.55\textwidth}
\begin{itemize}
\item Input-dependent noise modeled via heteroscedastic Gaussian process
\item Per-point noise variances set via \texttt{setNoise()}
\end{itemize}

\begin{lstlisting}[language=Python]
import openturns as ot
x = ot.Uniform(0.0, 10.0).getSample(20)
model = ot.SymbolicFunction("x", "x*sin(x)")
y = model(x)
noiseVar = ot.Uniform(0.0, 1.5).getSample(20).asPoint()
cov = ot.MaternModel([1.0], 1.5)
fitter = ot.GaussianProcessFitter(x, y, cov...)
fitter.setNoise(noiseVar)
fitter.run()
result = fitter.getResult()
\end{lstlisting}
\end{column}

\begin{frame}[fragile]
\frametitle{Cross-validation for GPR}

\begin{itemize}
\item Leave-one-out CV residuals computed via closed-form formulas
\item No refitting required
\end{itemize}

\begin{block}{LOO residual formula}
\[
e_{-i} = \frac{ [\matcov^{-1} \vect{y}]_i }{ [\matcov^{-1}]_{ii} }
\qquad
\sigma^2_{-i} = \frac{1}{[\matcov^{-1}]_{ii}}
\]
\small
Ginsbourger, Schärer (2025)
\end{block}

\begin{lstlisting}[language=Python]
fitter = ot.GaussianProcessFitter(x, y, covarianceModel, basis)
fitter.run()

cv = otexp.GaussianProcessRegressionCrossValidation(fitter.getResult())
cv.run()
result = cv.getResult()
r2 = cv.computeR2Score()
stddev = cv.getLeaveOneOutStandardDeviations()
\end{lstlisting}

\end{frame}

\begin{frame}[fragile]
\frametitle{HiGHS linear solver}

\begin{columns}
\begin{column}{0.55\textwidth}
\begin{block}{Open-source high-performance LP solver}
\begin{itemize}
\item Mixed-integer linear programming via HiGHS
\item Constraints: $L \leq A x \leq U$
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.4\textwidth}
\includegraphics[width=\textwidth]{figures/highs_banner}
\end{column}
\end{columns}

\begin{columns}
\begin{column}{0.45\textwidth}
\begin{lstlisting}[language=Python]
# Maximize x + y + 2z: x + 2y + 3z <= 4, x + y >= 1, x y z binary
cost = [1, 1, 2]
A = ot.Matrix([[1, 2, 3], [1, 1, 0]])
LU = ot.Interval([-1e30, 1], [4, 1e30])
bounds = ot.Interval(3)
problem = LinearProblem(cost, bounds, A, LU)
problem.setVariablesType([BINARY] * 3)
problem.setMinimization(False)
algo = HiGHS(problem)
algo.setStartingPoint([0.0] * 3)
algo.run()
result = algo.getResult()
x = result.getOptimalPoint()
y = result.getOptimalValue()
\end{lstlisting}
\end{column}

\begin{frame}[fragile]
\frametitle{FiniteOrthogonalFunctionFactory}

\begin{block}{Custom orthonormal basis}
\begin{itemize}
\item Build orthonormal basis from arbitrary functions
\item Gram-Schmidt orthogonalization wrt any measure
\end{itemize}
\end{block}

\begin{lstlisting}[language=Python, basicstyle=\footnotesize\ttfamily]
dist = distribution = ot.Normal(mean, std, R)
f0 = ot.SymbolicFunction(["x0", "x1"], ["1.0"])
f1 = ot.SymbolicFunction(["x0", "x1"], ["x0"])
f2 = ot.SymbolicFunction(["x0", "x1"], ["-4.0*x0/3.0+5.0*x1/3.0"])
f3 = ot.SymbolicFunction(["x0", "x1"], ["-1.0/sqrt(2.0)+x0^2/sqrt(2)"])
factory = otexp.FiniteOrthogonalFunctionFactory([f0, f1, f2, f3], dist)

prodB = ot.OrthogonalBasis(factory)
bSize = len(factory.getFunctionsCollection())
algo = ot.LeastSquaresExpansion(X, Y, distribution, prodB, bSize)
\end{lstlisting}

\end{frame}

e1 = ot.FejerExperiment(ot.Uniform(0.0, 1.0))
e2 = ot.FejerExperiment(ot.Uniform(0.0, 1.0))
smolyak = ot.SmolyakExperiment([e1, e2], 3)
nodes, weights = smolyak.generateWithWeights()

\begin{frame}[fragile]
\frametitle{MultiStart in FORM}

\begin{block}{Global design point search}
\begin{itemize}
\item \texttt{MultiStart} solver accepted by FORM/SORM
\item Runs optimizer from multiple starting points
\item Avoids local minima in design point computation
\end{itemize}
\end{block}

\begin{lstlisting}[language=Python]
import openturns as ot
dist = ot.Normal(2)
event = ot.ThresholdEvent(ot.RandomVector(dist), ot.Greater(), 1.0)
start = dist.getSample(10)
optim = ot.MultiStart(ot.Cobyla(), start)
algo = ot.FORM(optim, event)
algo.run()
result = algo.getResult()
\end{lstlisting}

\end{frame}

\begin{frame}[fragile]
\frametitle{QuantileConfidence interval coverage}

\begin{block}{Actual coverage probability}
\begin{itemize}
\item Quantile $x_\alpha$: $P(X \leq x_\alpha) = \alpha$
\item Confidence level: $P(x_\alpha \in \text{CI}) \ge \beta$
\item New \texttt{*WithCoverage} methods return true $\beta$
\end{itemize}
\end{block}

\begin{lstlisting}[language=Python]
import openturns as ot
sample = ot.Normal().getSample(100)
alpha = 0.9
beta = 0.95
algo = ot.QuantileConfidence(alpha, beta)
ci, cov = algo.computeUnilateralConfidenceIntervalWithCoverage(sample)
\end{lstlisting}

\end{frame}

\begin{frame}[fragile]
\frametitle{KFoldSplitter non-interlaced folds}

\begin{block}{Contiguous fold blocks}
\begin{itemize}
\item Folds are contiguous (non-interlaced) index ranges
\item Each fold: $[0, N/k), [N/k, 2N/k), \ldots$
\item Optional randomization via \texttt{setRandomize(True)}
\end{itemize}
\end{block}

\begin{lstlisting}[language=Python]
import openturns as ot
sample = ot.Normal().getSample(10)
splitter = ot.KFoldSplitter(sample.getSize(), 5)
for train, test in splitter:
    trainSample = sample[train]
    testSample = sample[test]
\end{lstlisting}

\end{frame}

\begin{frame}[fragile]
\frametitle{KDTree radius search}

\begin{block}{Range queries with nanoflann}
\begin{itemize}
\item New \texttt{queryRadius} method for range search
\item Returns all points within a given radius
\item Optional sorted output by distance
\end{itemize}
\end{block}

\begin{lstlisting}[language=Python]
import openturns as ot
sample = ot.Normal(3).getSample(100)
tree = ot.KDTree(sample)
indices, dist = tree.queryRadius([0.0] * 3, 0.5, True)
\end{lstlisting}

\end{frame}

\begin{frame}[fragile]
\frametitle{Pagmo incremental evolution callback}

\begin{columns}
\begin{column}{0.55\textwidth}
\begin{block}{Per-generation progress reporting}
\begin{itemize}
\item Evolve one generation at a time instead of all at once
\item Continuously report population and Pareto fronts
\item Enabled via \texttt{setIncrementalEvolution(True)}
\end{itemize}
\end{block}

\begin{lstlisting}[language=Python]
problem = ot.OptimizationProblem(f)
algo = ot.Pagmo(problem, "nsga2", x0)
algo.setIncrementalEvolution(True)
algo.setProgressCallback(callback)
algo.run()
result = algo.getResult()
